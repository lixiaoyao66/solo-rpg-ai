## 🗄️ 2. 数据库设计

### 2.1 数据库表结构

#### users 表
```sql
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email VARCHAR(255) UNIQUE NOT NULL,
  username VARCHAR(50) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  avatar_url TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  last_login_at TIMESTAMPTZ
);

-- 索引
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_username ON users(username);
```

#### characters 表
```sql
CREATE TABLE characters (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  
  -- 基础信息
  name VARCHAR(100) NOT NULL,
  race VARCHAR(50) NOT NULL,
  class VARCHAR(50) NOT NULL,
  level INTEGER DEFAULT 1,
  background TEXT,
  
  -- 基础属性
  strength INTEGER NOT NULL,
  dexterity INTEGER NOT NULL,
  constitution INTEGER NOT NULL,
  intelligence INTEGER NOT NULL,
  wisdom INTEGER NOT NULL,
  charisma INTEGER NOT NULL,
  
  -- 战斗数据
  hp_current INTEGER NOT NULL,
  hp_max INTEGER NOT NULL,
  ac INTEGER NOT NULL,
  initiative INTEGER,
  speed INTEGER NOT NULL,
  
  -- 技能和熟练（JSON）
  skills JSONB NOT NULL DEFAULT '{}',
  saving_throws JSONB NOT NULL DEFAULT '{}',
  
  -- 法术和能力（JSON）
  spells JSONB DEFAULT '[]',
  special_abilities JSONB DEFAULT '[]',
  
  -- 装备和背包（JSON）
  equipment JSONB DEFAULT '[]',
  inventory JSONB DEFAULT '[]',
  gold INTEGER DEFAULT 0,
  
  -- 状态
  conditions JSONB DEFAULT '[]',
  
  -- 背景故事
  backstory TEXT,
  personality TEXT,
  goals JSONB DEFAULT '[]',
  
  -- 立绘
  portrait_url TEXT,
  
  -- 元数据
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_characters_user_id ON characters(user_id);
CREATE INDEX idx_characters_name ON characters(name);
```

#### campaigns 表
```sql
CREATE TABLE campaigns (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  character_id UUID NOT NULL REFERENCES characters(id) ON DELETE CASCADE,
  
  -- 基本信息
  title VARCHAR(200) NOT NULL,
  description TEXT,
  setting TEXT, -- 世界观设定
  
  -- 游戏状态
  status VARCHAR(20) DEFAULT 'active' CHECK (status IN ('active', 'completed', 'paused')),
  current_chapter INTEGER DEFAULT 1,
  
  -- 世界状态（JSON）
  world_state JSONB DEFAULT '{
    "currentLocation": "",
    "activeQuests": [],
    "completedQuests": [],
    "npcRelationships": {},
    "majorDecisions": []
  }',
  
  -- 记忆数据（JSON）
  memory JSONB DEFAULT '{
    "sessionSummaries": [],
    "keyEvents": []
  }',
  
  -- 元数据
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  last_played_at TIMESTAMPTZ DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_campaigns_user_id ON campaigns(user_id);
CREATE INDEX idx_campaigns_character_id ON campaigns(character_id);
CREATE INDEX idx_campaigns_status ON campaigns(status);
```

#### game_sessions 表
```sql
CREATE TABLE game_sessions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  campaign_id UUID NOT NULL REFERENCES campaigns(id) ON DELETE CASCADE,
  
  -- 会话摘要
  summary TEXT,
  key_events JSONB DEFAULT '[]',
  
  -- 状态变化
  character_changes JSONB DEFAULT '{
    "hpChange": 0,
    "xpGained": 0,
    "levelUp": false,
    "newItems": []
  }',
  
  -- 元数据
  started_at TIMESTAMPTZ DEFAULT NOW(),
  ended_at TIMESTAMPTZ
);

-- 索引
CREATE INDEX idx_game_sessions_campaign_id ON game_sessions(campaign_id);
CREATE INDEX idx_game_sessions_started_at ON game_sessions(started_at);
```

#### messages 表
```sql
CREATE TABLE messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  session_id UUID NOT NULL REFERENCES game_sessions(id) ON DELETE CASCADE,
  
  -- 消息内容
  role VARCHAR(20) NOT NULL CHECK (role IN ('ai', 'player', 'system')),
  content TEXT NOT NULL,
  
  -- 骰子相关（JSON）
  dice_roll JSONB,
  -- 结构：{
  --   "type": "1d20",
  --   "results": [15],
  --   "total": 15,
  --   "modifier": 3,
  --   "checkType": "skill",
  --   "dc": 15,
  --   "success": true
  -- }
  
  -- 选项（AI消息）
  options JSONB DEFAULT '[]',
  
  -- 元数据
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_messages_session_id ON messages(session_id);
CREATE INDEX idx_messages_role ON messages(role);
CREATE INDEX idx_messages_created_at ON messages(created_at);
```

#### saves 表（存档）
```sql
CREATE TABLE saves (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  campaign_id UUID NOT NULL REFERENCES campaigns(id) ON DELETE CASCADE,
  session_id UUID REFERENCES game_sessions(id) ON DELETE SET NULL,
  
  -- 存档信息
  name VARCHAR(100) NOT NULL,
  description TEXT,
  save_type VARCHAR(20) DEFAULT 'manual' CHECK (save_type IN ('manual', 'auto')),
  
  -- 存档数据（JSON）
  campaign_snapshot JSONB NOT NULL,
  character_snapshot JSONB NOT NULL,
  world_state_snapshot JSONB NOT NULL,
  
  -- 元数据
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_saves_campaign_id ON saves(campaign_id);
CREATE INDEX idx_saves_save_type ON saves(save_type);
```

#### assets 表（资源管理）
```sql
CREATE TABLE assets (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  
  -- 资源信息
  asset_type VARCHAR(50) NOT NULL CHECK (asset_type IN ('character_portrait', 'monster', 'npc', 'scene')),
  name VARCHAR(200) NOT NULL,
  description TEXT,
  
  -- 存储
  url TEXT NOT NULL,
  thumbnail_url TEXT,
  
  -- 来源
  source VARCHAR(20) DEFAULT 'ai' CHECK (source IN ('ai', 'upload', 'preset')),
  prompt TEXT, -- AI生成时的提示词
  
  -- 元数据
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_assets_user_id ON assets(user_id);
CREATE INDEX idx_assets_asset_type ON assets(asset_type);
```

### 4.1 AI API 选择

### 4.3 Prompt 设计

#### System Prompt（核心）
```
你是一个DND 5e地下城主（DM），负责主持一场单人冒险。

## 你的职责
1. 描述场景和环境
2. 扮演NPC和怪物
3. 根据DND 5e规则判断何时需要检定
4. 根据检定结果描述后果
5. 推进故事情节

## 规则遵循
- 战斗中的行动严格按DND 5e规则
- 攻击需要攻击骰，法术需要法术攻击或豁免
- 探索/社交中，只有"有意义的风险"才需要检定
- 明确告知玩家检定类型和DC

## 检定结果分级
- 大成功（自然20）：超额完成 + 额外好处
- 成功：达成目标
- 部分失败（差1-2点）：达成目标但有代价
- 失败：未达成目标
- 大失败（自然1）：灾难性后果

## 响应格式
每次响应必须包含：
1. 场景/结果描述（叙事性）
2. 如果需要检定，明确说明检定类型和DC
3. 提供2-4个行动选项（可选）

## 当前游戏状态
{game_state}

## 角色信息
{character_info}

## 世界状态
{world_state}

## 记忆摘要
{memory_summary}
```

#### 怪物AI Prompt
```
你正在控制一个怪物进行战斗。

## 怪物信息
- 名称：{monster_name}
- HP：{current_hp}/{max_hp}
- 性格：{personality}
- 战术倾向：{tactics}

## 战斗状态
- 玩家HP：{player_hp}
- 玩家AC：{player_ac}
- 玩家位置：{player_position}
- 盟友状态：{allies_status}

## 决策规则
根据性格和当前状态做出决策：
- 如果 HP < 50% 且性格为"胆小"，考虑逃跑
- 如果性格为"好战"，战斗到最后
- 如果性格为"狡猾"，优先攻击弱势目标

## 输出格式
{
  "action": "attack" | "move" | "cast" | "flee" | ...,
  "target": "player" | "ally" | ...,
  "description": "描述怪物的行动"
}
```

### 4.5 上下文构建
```python
def build_context(campaign: Campaign, character: Character) -> str:
    context = []
    
    # 1. 角色信息（始终包含）
    context.append(f"## 角色信息\n{character.to_prompt()}")
    
    # 2. 当前世界状态
    context.append(f"## 当前状态\n{campaign.world_state.to_prompt()}")
    
    # 3. 记忆摘要（按优先级）
    memory = campaign.memory
    
    # 3.1 关键事件（最重要）
    if memory.key_events:
        context.append("## 关键事件\n" + "\n".join(memory.key_events[-5:]))
    
    # 3.2 最近的会话摘要
    if memory.session_summaries:
        recent = memory.session_summaries[-3:]
        context.append("## 最近冒险\n" + "\n".join([s.summary for s in recent]))
    
    # 3.3 长期记忆
    context.append(f"## 世界记忆\n{memory.to_prompt()}")
    
    return "\n\n".join(context)
```

### 6.2 权限控制
```python
# 依赖注入
async def get_current_user(
    token: str = Depends(oauth2_scheme),
    db: AsyncSession = Depends(get_db)
) -> User:
    # 验证 token
    payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
    user_id = payload.get("sub")
    
    # 查询用户
    user = await db.get(User, user_id)
```

```python
from pydantic import BaseModel, validator

class CharacterCreate(BaseModel):
    name: str = Field(..., min_length=1, max_length=100)
    race: str
    class_: str = Field(..., alias="class")
    abilities: Dict[str, int]
    
    @validator('abilities')

        required = ['strength', 'dexterity', 'constitution', 
                   'intelligence', 'wisdom', 'charisma']
```
